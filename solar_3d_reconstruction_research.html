const fs = require('fs');
async function dl(url, dest) {
  fs.mkdirSync('models_3d', { recursive: true });
  const res = await fetch(url);
  const buf = await res.arrayBuffer();
  fs.writeFileSync(dest, Buffer.from(buf));
  console.log('Descargado:', dest);
}
